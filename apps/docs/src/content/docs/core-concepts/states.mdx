## Basic states

Examples build on each other: paste them into one file, in order.

```ts
import { state } from "@mantaq/core";

const basicInfo = state("basicInfo")();
const submitting = state("submitting")();
basicInfo.name; // "basicInfo"
basicInfo.isFinal; // false
```

## Final states

```ts
const success = state("success")().final();
success.isFinal; // true
```

A final state fires `"done"`, sets `snapshot().done`, and stops the actor.

## Payloads

```ts
import { Actor, event } from "@mantaq/core";

const error = state("error")<{ reason: string }>();
const paymentFail = event("paymentFail")<{ reason: string }>();

const payments = Actor({
  inputs: [paymentFail],
  states: [submitting, error],
  initial: submitting,
  setup: (m) => {
    m.on(submitting, {
      event: paymentFail,
      to: [error],
      handler: (event) => ({ state: error, payload: { reason: event.payload.reason } }),
    });
    m.effect(error, {
      name: "logFailure",
      fn: ({ state }) => {
        console.log(state.payload.reason);
      },
    });
  },
});

payments.send(paymentFail.create({ reason: "card declined" })); // logs "card declined"
```

## Regions

```ts
import { matches } from "@mantaq/sugar";

const scanning = state("scanning")();
const cleared = state("cleared")();
const flagged = state("flagged")();
const scanDone = event("scanDone")<{ risky: boolean }>();

const fraudMonitor = Actor({
  inputs: [scanDone],
  states: [scanning, cleared, flagged],
  initial: scanning,
  setup: (m) => {
    m.on(scanning, {
      event: scanDone,
      to: [cleared, flagged],
      handler: (event) => ({ state: event.payload.risky ? flagged : cleared }),
    });
  },
});

const checkout = Actor({
  inputs: [],
  states: [basicInfo],
  initial: basicInfo,
  setup: () => {},
  regions: { fraud: fraudMonitor },
});

fraudMonitor.send(scanDone.create({ risky: false }));
matches(checkout, "basicInfo.fraud.cleared"); // true
```

Each region is its own actor.

---
title: Matching
description: State matching, grouping, and leaf detection.
---

Check where actor is. Group states. Find leaves.

## matches

Dot-notation path check. Prefix match.

```ts
import { matches } from "@mantaq/sugar";

matches(actor, "active"); // top-level
matches(actor, "connected.default.active"); // nested path
```

Starts at the top-level state. Descends only through region names, in order. Ends at a state or region boundary. NOT a whole-tree search — a nested `"active"` under a different top state fails `matches(actor, "active")`. Use `isIn` for that.

Without sugar:

```ts
actor.snapshot().path[0] === "active";
// or for nested:
const snap = actor.snapshot();
snap.path[0] === "connected" && snap.regions["default"]?.path[0] === "active";
```

`matches` handles path joining.

## isIn

Recursive search. State anywhere in the snapshot tree.

```ts
import { isIn } from "@mantaq/sugar";

isIn(actor.snapshot(), "active"); // true if "active" anywhere in the tree
```

Useful with hierarchical states:

```ts
// snapshot: { path: ["connected"], regions: { default: { path: ["active"], regions: {} } } }
isIn(snapshot, "connected"); // true
isIn(snapshot, "active"); // true
isIn(snapshot, "idle"); // false
```

Without sugar:

```ts
import type { Snapshot } from "@mantaq/core";

function inTree(snap: Snapshot, name: string): boolean {
  if (snap.path[0] === name) return true;
  return Object.values(snap.regions).some((r) => inTree(r, name));
}
inTree(actor.snapshot(), "active");
```

## activeLeaves

All leaf states in current path.

```ts
import { activeLeaves } from "@mantaq/sugar";

// snapshot.path: ["connected"], regions: { default: { path: ["active"] } }
activeLeaves(actor.snapshot()); // ["connected.default.active"]
```

For parallel regions, returns multiple leaves.

Without sugar:

```ts
// Manual. More code. Sugar handles edge cases.
```

## tag

Group states. Check membership.

```ts
import { tag } from "@mantaq/sugar";

const activeTag = tag(idle, loading, success);
activeTag.has(actor.snapshot()); // true if any of those states anywhere in the tree
```

Use for bulk checks:

```ts
const errorStates = tag(networkError, validationError, timeoutError);

if (errorStates.has(actor.snapshot())) {
  showErrorUI();
}
```

Without sugar:

```ts
const path = actor.snapshot().path[0];
if (path === "networkError" || path === "validationError" || path === "timeoutError") {
  showErrorUI();
}
```

Top-level only. `tag.has` searches the whole tree.

Tag scales better. Many states → one check.

## Before/after

### Check state

```ts
// Before (core only)
actor.snapshot().path[0] === "active";

// After (sugar)
matches(actor, "active");
```

### Check nested state

```ts
// Before
import type { Snapshot } from "@mantaq/core";

function inTree(snap: Snapshot, name: string): boolean {
  if (snap.path[0] === name) return true;
  return Object.values(snap.regions).some((r) => inTree(r, name));
}
inTree(actor.snapshot(), "active");

// After
isIn(actor.snapshot(), "active");
```

### Group errors

```ts
import { state } from "@mantaq/core";
import { tag } from "@mantaq/sugar";

const err1 = state("err1")();
const err2 = state("err2")();
const err3 = state("err3")();

// Before
const isError = [err1, err2, err3].some((s) => s.name === actor.snapshot().path[0]);

// After
const isError = tag(err1, err2, err3).has(actor.snapshot());
```

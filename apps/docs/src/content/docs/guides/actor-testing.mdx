Actors are state machines. States are nodes, transitions are edges. Testing means walking the graph and asserting you land where expected.

## Simple Example: Traffic Light

States: `green` → `yellow` → `red` → `green` (cycle)
Events: `NEXT`

```ts
import { Actor, state, event } from "@mantaq/core";

const green = state("green")();
const yellow = state("yellow")();
const red = state("red")();

const next = event("NEXT")();

function createTrafficLight() {
  return new Actor({
    inputs: [next],
    states: [green, yellow, red],
    initial: green,
    context: {},
    setup: (m) => {
      m.on(green, next, () => ({ state: yellow }));
      m.on(yellow, next, () => ({ state: red }));
      m.on(red, next, () => ({ state: green }));
    },
  });
}
```

## Testing Patterns

### Assert Initial State

```ts
const actor = createTrafficLight();
expect(actor.snapshot().path[0]).toBe("green");
expect(actor.snapshot().done).toBeFalsy();
```

### Assert Transition Path

```ts
actor.send(next.create());
expect(actor.snapshot().path[0]).toBe("yellow");

actor.send(next.create());
expect(actor.snapshot().path[0]).toBe("red");

actor.send(next.create());
expect(actor.snapshot().path[0]).toBe("green"); // cycle back
```

### Assert Context Changes

If your actor stores data in context:

```ts
const counter = state("counter")();
const increment = event("INCREMENT")();

const actor = new Actor({
  inputs: [increment],
  states: [counter],
  initial: counter,
  context: { count: 0 },
  setup: (m) => {
    m.on(counter, increment, (_event, { context }) => {
      const s = context.get();
      context.set({ ...s, count: s.count + 1 });
      return {};
    });
  },
});

actor.send(increment.create());
expect(actor.context.count).toBe(1);
```

### Assert Side Effects with VirtualClock

For actors with timers or async effects:

```ts
import { Actor, state, event, VirtualClock } from "@mantaq/core";

const idle = state("idle")();
const loading = state("loading")();
const done = state("done")();

const start = event("START")();
const finish = event("FINISH")();

const clock = new VirtualClock();

const actor = new Actor({
  inputs: [start],
  internal: [finish],
  states: [idle, loading, done],
  initial: idle,
  clock,
  setup: (m) => {
    m.on(idle, start, () => ({ state: loading }));
    m.effect(loading, ({ signal, emit, clock }) => {
      clock.setTimeout(1000, () => emit(finish.create()), { signal });
    });
    m.on(loading, finish, () => ({ state: done }));
  },
});

actor.send(start.create());
expect(actor.snapshot().path[0]).toBe("loading");

clock.advance(1000); // fast-forward time
expect(actor.snapshot().path[0]).toBe("done");
```

### Assert Final State

```ts
const idle = state("idle")();
const success = state("success")().final();
const finish = event("FINISH")();

const actor = new Actor({
  inputs: [finish],
  states: [idle, success],
  initial: idle,
  setup: (m) => {
    m.on(idle, finish, () => ({ state: success }));
  },
});

actor.send(finish.create());
expect(actor.snapshot().done).toBe(true);
```

## More Complex Example: Login Flow

States: `idle` → `authenticating` → `authenticated` | `error`
Events: `LOGIN`, `LOGOUT`, `RETRY`, `AUTH_DONE`, `AUTH_FAIL`

```ts
const idle = state("idle")();
const authenticating = state("authenticating")();
const authenticated = state("authenticated")();
const error = state("error")();

const login = event("LOGIN")<{ username: string; password: string }>();
const logout = event("LOGOUT")();
const retry = event("RETRY")();
const authDone = event("AUTH_DONE")();
const authFail = event("AUTH_FAIL")();

type AuthContext = { username?: string; attempts: number };

function createLoginActor() {
  return new Actor({
    inputs: [login, logout, retry, authDone, authFail],
    states: [idle, authenticating, authenticated, error],
    initial: idle,
    context: { attempts: 0 } as AuthContext,
    setup: (m) => {
      m.on(idle, login, (event, { context }) => {
        const s = context.get();
        context.set({ ...s, username: event.payload.username, attempts: s.attempts + 1 });
        return { state: authenticating };
      });
      m.on(authenticating, authDone, () => ({ state: authenticated }));
      m.on(authenticating, authFail, () => ({ state: error }));
      m.on(error, retry, () => ({ state: idle }));
      m.on(authenticated, logout, (_event, { context }) => {
        const s = context.get();
        context.set({ ...s, username: undefined });
        return { state: idle };
      });
    },
  });
}
```

### Test the Full Flow

```ts
describe("login actor", () => {
  it("idle → authenticating → authenticated", () => {
    const actor = createLoginActor();
    expect(actor.snapshot().path[0]).toBe("idle");

    actor.send(login.create({ username: "user", password: "pass" }));
    expect(actor.snapshot().path[0]).toBe("authenticating");

    actor.send(authDone.create());
    expect(actor.snapshot().path[0]).toBe("authenticated");
    expect(actor.context.username).toBe("user");
  });

  it("idle → authenticating → error → retry → idle", () => {
    const actor = createLoginActor();

    actor.send(login.create({ username: "user", password: "wrong" }));
    actor.send(authFail.create());
    expect(actor.snapshot().path[0]).toBe("error");

    actor.send(retry.create());
    expect(actor.snapshot().path[0]).toBe("idle");
  });

  it("authenticated → logout → idle clears username", () => {
    const actor = createLoginActor();

    actor.send(login.create({ username: "user", password: "pass" }));
    actor.send(authDone.create());
    expect(actor.snapshot().path[0]).toBe("authenticated");

    actor.send(logout.create());
    expect(actor.snapshot().path[0]).toBe("idle");
    expect(actor.context.username).toBeUndefined();
  });
});
```

## Key Takeaways

- **States are nodes** — `state("name")()` creates a node in the graph
- **Events are edges** — `event("ID")()` defines a possible transition
- **Transitions are the graph** — `(state, event) → newState` maps edges
- **Context is memory** — survives transitions, holds accumulated data
- **Effects are side effects** — run on state entry, abort on exit
- **`snapshot().path[0]`** — check current state name in graph
- **`VirtualClock`** — deterministically test time-dependent transitions
- **`snapshot().done`** — check if actor reached a final state
